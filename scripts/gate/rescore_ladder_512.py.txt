"""8x8 per-cell coverage re-score for the 512 gate-precision ladder (L1/L2/L3).

Generalized sibling of ``rescore_v0_10_percell_coverage.py`` (§8 of
``documents/history/specs/stage1_gate_precision_ladder_512_spec.md``). Same
INFERENCE-ONLY diagnostic (no training, no backward, no checkpoint/code-path
mutation): forward-passes each arm's saved ``best_gate_model.pth`` over the
exact frozen val set / threshold 0.5 / native low-res grid (32x128x128) and
scores coverage on the 8x8 = 64-cell grid (16x16 prediction-voxel blocks,
time-collapsed over the 32 frames).

Differences from the v0.10 script (per §8 a/b/c):
  (a) ARMS / per-arm build args are parameterized -- no v0.10-specific
      GATE_EPOCH / LOGGED_CLUSTER_RECALL / B1 anchor.
  (b) ``stem_stride`` and ``padding_mode`` are set PER ARM to match each
      checkpoint's architecture (L1 -> stem_stride=2; L2/L3 -> 1; all three
      -> padding_mode='replicate'), so the weight load and forward both
      reproduce the trained model exactly.
  (c) runs layout expected: ``<runs_root>/<arm>/models/best_gate_model.pth``.

Report (§8), per arm, at k=1 (plus the full k-sweep for context):
  * per-cell FN = 1 - strict cell recall on target cells, vs goal 1.0%;
  * absent-scene FP = mean cells flagged / 64, vs goal 25.0%;
  * present-scene non-target FP mean + p50/p95 tail (the flood is bimodal);
  * marginal deltas (L1-baseline, L2-L1, L3-L2) are assembled by the caller
    from this JSON + the on-disk rescore.json baseline (Arm A ep42).

Needs a CUDA GPU for practical speed (Mamba kernels); data/run/checkpoint
locations resolve under $LSL_DATA_ROOT / $LSL_RUNS_ROOT / $LSL_CKPT_ROOT:
  python scripts/gate/rescore_ladder_512.py --runs_dir <runs>/stage1_gate_ladder_512 --out rescore.json
"""

from __future__ import annotations

import argparse
import json
import socket
import sys
from pathlib import Path

import numpy as np
import torch



from localizer.gate.model import build_segmenter_v0_7
from localizer.common.checkpoint_io import load_state_dict
from localizer.common.paths import add_root_args, resolve_under
from localizer.common.data import make_dataloaders

# Frozen eval constants (identical across L1/L2/L3 -- §6).
LOW_RES_POOL = (2, 4, 4)
SUBSET_SIZE = 3000
VAL_SPLIT = 0.1
VAL_SPLIT_SEED = 0
THRESHOLD = 0.5
CELL = 16            # 16x16 prediction-voxel blocks
NGRID = 8            # 8x8 = 64 cells over the 128x128 native grid
K_SWEEP = (1, 2, 4, 8)

# Operator goals (§1) for inline comparison in the printout.
GOAL_FN = 0.01       # per-cell FN < 1.0%
GOAL_FP = 0.25       # absent-scene FP < 25.0%

# Per-arm build args. stem_stride MUST match the trained checkpoint
# (§8b); padding_mode MUST match so the forward reproduces training
# (border fix is replicate for all three rungs).
ARM_BUILD = {
    "L0": dict(arm=2, nfeats=32, fusion="sum", stem_stride=2,
               dt_init_std=None, padding_mode="zeros"),
    "L1": dict(arm=2, nfeats=32, fusion="sum", stem_stride=2,
               dt_init_std=None, padding_mode="replicate"),
    "L2": dict(arm=2, nfeats=32, fusion="sum", stem_stride=1,
               dt_init_std=None, padding_mode="replicate"),
    "L3": dict(arm=2, nfeats=32, fusion="sum", stem_stride=1,
               dt_init_std=None, padding_mode="replicate"),
}
DEFAULT_ARMS = ("L1", "L2", "L3")


def build_val_dataset(data_root: str):
    """Reconstruct the exact val membership the live pipeline used.

    ``make_dataloaders`` fixes val membership deterministically via
    ``train_test_split(random_state=VAL_SPLIT_SEED)`` -- independent of the
    loader shuffle. We take the val_loader's underlying dataset and iterate
    it in fixed index order (no shuffle) for reproducibility.
    """
    _, val_loader, datshape = make_dataloaders(
        data_root=data_root, subset_size=SUBSET_SIZE, batch_size=1, num_workers=0,
        val_split=VAL_SPLIT, val_split_seed=VAL_SPLIT_SEED,
        output_resolution="low_res", soft_target=False, low_res_pool=LOW_RES_POOL,
    )
    return val_loader.dataset, datshape


def cell_counts(voxel_bool: np.ndarray) -> np.ndarray:
    """Collapse a (T, 128, 128) boolean voxel grid to an (8, 8) per-cell COUNT.

    Count = number of True voxels in each 16x16 spatial block summed across
    all T frames. ``any`` == (count >= 1).
    """
    t, h, w = voxel_bool.shape
    assert h == NGRID * CELL and w == NGRID * CELL, f"unexpected spatial {h}x{w}"
    r = voxel_bool.reshape(t, NGRID, CELL, NGRID, CELL)
    return r.sum(axis=(0, 2, 4)).astype(np.int64)   # (8, 8)


@torch.no_grad()
def predict_voxel_pos(model, volume_chw_t: torch.Tensor, device) -> np.ndarray:
    """Forward one scene; return (T,128,128) bool of sigmoid(logits)>0.5 (native)."""
    x = volume_chw_t.unsqueeze(0).to(device)             # (1,1,64,512,512)
    logits = model(x).float()                            # (1,1,32,128,128)
    pos = (torch.sigmoid(logits) > THRESHOLD)            # bool
    return pos[0, 0].cpu().numpy()                       # (32,128,128)


def load_model(ckpt_path: Path, build: dict, device):
    """Build with the arm's exact build args and strict-load the checkpoint."""
    model = build_segmenter_v0_7(**build).to(device)
    sd = load_state_dict(ckpt_path, map_location=device)
    if isinstance(sd, dict) and "state_dict" in sd and "model" not in sd:
        sd = sd["state_dict"]
    model.load_state_dict(sd, strict=True)
    model.eval()
    return model


def score_arm(arm: str, runs_root: Path, dataset, device,
              ckpt_name: str = "best_gate_model.pth") -> dict:
    ckpt = runs_root / arm / "models" / ckpt_name
    if not ckpt.is_file():
        raise FileNotFoundError(f"missing checkpoint for arm {arm}: {ckpt}")
    build = ARM_BUILD[arm]
    model = load_model(ckpt, build, device)

    n = len(dataset)
    present_caught = {k: 0 for k in K_SWEEP}
    present_truth_cells_tot = 0                              # k-independent
    present_truthpred_cells = {k: 0 for k in K_SWEEP}
    present_nontarget_flagged = {k: [] for k in K_SWEEP}     # per-scene
    absent_flagged = {k: [] for k in K_SWEEP}               # per-scene
    n_present = n_absent = 0

    for i in range(n):
        s = dataset[i]
        label = int(s["label"])
        vol = torch.as_tensor(s["volume"])                  # (1,64,512,512)
        truth_vox = (np.asarray(s["mask_hard"])[0] > 0.5)   # (32,128,128) bool
        pred_vox = predict_voxel_pos(model, vol, device)    # (32,128,128) bool

        truth_cnt = cell_counts(truth_vox)                  # (8,8) int
        pred_cnt = cell_counts(pred_vox)                    # (8,8) int
        truth_cells = truth_cnt >= 1                        # any-voxel truth rule

        if label == 1:
            n_present += 1
            present_truth_cells_tot += int(truth_cells.sum())
        else:
            n_absent += 1

        for k in K_SWEEP:
            pred_cells = pred_cnt >= k
            if label == 1:
                overlap = truth_cells & pred_cells
                if overlap.any():
                    present_caught[k] += 1
                present_truthpred_cells[k] += int(overlap.sum())
                present_nontarget_flagged[k].append(
                    int((pred_cells & ~truth_cells).sum())
                )
            else:
                absent_flagged[k].append(int(pred_cells.sum()))

    def pdict(vals):
        a = np.asarray(vals, dtype=np.float64) if vals else np.array([0.0])
        return dict(mean=float(a.mean()), p50=float(np.percentile(a, 50)),
                    p95=float(np.percentile(a, 95)), max=float(a.max()))

    per_k = {}
    for k in K_SWEEP:
        cell_recall = (
            present_truthpred_cells[k] / max(present_truth_cells_tot, 1)
        )
        per_k[str(k)] = {
            "caught_rate": present_caught[k] / max(n_present, 1),
            "per_cell_recall_strict": cell_recall,
            "per_cell_fn": 1.0 - cell_recall,
            "absent_cells_flagged": pdict(absent_flagged[k]),
            "absent_cells_flagged_rate64": pdict(
                [v / 64.0 for v in absent_flagged[k]]
            ),
            "present_nontarget_cells_flagged": pdict(present_nontarget_flagged[k]),
        }

    return {
        "arm": arm, "ckpt": str(ckpt), "build": build,
        "n_present": n_present, "n_absent": n_absent,
        "present_truth_cells_total": present_truth_cells_tot,
        "per_k": per_k,
    }


RUNS_REQUIRED = True


def main() -> int:
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--dataset", default="multi_target_swarm_512x512x64",
                   help="dataset name under the data root, or an absolute path")
    p.add_argument("--runs_dir", required=RUNS_REQUIRED,
                   help="run directory holding <arm>/models/*.pth; relative to the runs root, or absolute")
    p.add_argument("--out", required=True)
    p.add_argument(
        "--arms", default=",".join(DEFAULT_ARMS),
        help="comma-separated arm names (default L1,L2,L3)",
    )
    p.add_argument(
        "--ckpt_name", default="best_gate_model.pth",
        help="checkpoint filename under <runs_root>/<arm>/models/ to score "
             "(default best_gate_model.pth; use best_stage1_seg_model_by_"
             "recall_at_15.pth to score the early-peak regime for arms that "
             "never produced a gate checkpoint)",
    )
    add_root_args(p, ("data", "runs", "ckpt"))
    args = p.parse_args()
    args.data_root = str(resolve_under("data", args.dataset, args.data_root_dir))
    args.runs_root = (str(resolve_under("runs", args.runs_dir, args.runs_root_dir))
                      if args.runs_dir else None)
    arms = tuple(a.strip() for a in args.arms.split(",") if a.strip())
    for a in arms:
        if a not in ARM_BUILD:
            raise SystemExit(f"unknown arm {a!r}; known: {sorted(ARM_BUILD)}")

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"device={device}  host={socket.gethostname()}")
    if device.type != "cuda":
        print("WARNING: no CUDA -- the Mamba kernel requires GPU; expect failure on CPU.")

    dataset, datshape = build_val_dataset(args.data_root)
    print(f"val scenes={len(dataset)}  datshape={datshape}")
    runs_root = Path(args.runs_root) if args.runs_root else None

    results = {}
    for arm in arms:
        print(f"--- scoring {arm} ({ARM_BUILD[arm]}) ---", flush=True)
        sc = score_arm(arm, runs_root, dataset, device, ckpt_name=args.ckpt_name)
        results[arm] = sc
        ck1 = sc["per_k"]["1"]
        fn = ck1["per_cell_fn"]
        fp = ck1["absent_cells_flagged_rate64"]["mean"]
        print(
            f"  {arm}: per_cell_FN@1={fn*100:.2f}% (goal {GOAL_FN*100:.1f}%) "
            f"absent_FP@1={fp*100:.2f}% (goal {GOAL_FP*100:.1f}%) "
            f"caught@1={ck1['caught_rate']:.3f} "
            f"presentNonTgtFP@1 mean={ck1['present_nontarget_cells_flagged']['mean']:.2f} "
            f"p95={ck1['present_nontarget_cells_flagged']['p95']:.1f}"
        )

    payload = {
        "meta": {
            "threshold": THRESHOLD, "grid": f"{NGRID}x{NGRID}",
            "cell_voxels": f"{CELL}x{CELL}", "k_sweep": list(K_SWEEP),
            "arm_build": ARM_BUILD, "ckpt_name": args.ckpt_name,
            "low_res_pool": list(LOW_RES_POOL),
            "val_split": VAL_SPLIT, "val_split_seed": VAL_SPLIT_SEED,
            "subset_size": SUBSET_SIZE, "data_root": args.data_root,
           , "goal_fn": GOAL_FN, "goal_fp": GOAL_FP,
        },
        "results": results,
    }
    outp = Path(args.out)
    outp.parent.mkdir(parents=True, exist_ok=True)
    outp.write_text(json.dumps(payload, indent=2, default=str))
    print(f"wrote {outp}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
