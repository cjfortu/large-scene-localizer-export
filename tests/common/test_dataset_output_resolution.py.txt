"""Tests for the v0.6 ``output_resolution`` flag on ``MoverDataset``.

Verifies:
* ``"full"`` mode preserves the v0/v0.5 return contract
  (``mask`` is the full-resolution mask, no ``mask_full`` key).
* ``"low_res"`` mode returns the max-pooled mask as ``mask``
  ``(1, 32, 32, 32)`` and the original full-resolution mask as
  ``mask_full`` ``(1, 64, 128, 128)``.
* The downsample matches a manual max-pool of the original mask
  (preserving "any positive pixel in the (2, 4, 4) region" semantics).
* Invalid ``output_resolution`` values raise ``ValueError``.

Uses a one-scene synthetic fixture (uint8 volume + sparse mask, the
dataset's on-disk format) written under ``tmp_path``; no external data.
"""

from __future__ import annotations

import os
from pathlib import Path

import numpy as np
import pytest
import torch
import torch.nn.functional as F

from localizer.common.data import MoverDataset


@pytest.fixture(scope="module")
def scene_dirs(tmp_path_factory) -> tuple[Path, Path, str]:
    """Write one 64x128x128 scene (uint8 volume + mask) in the on-disk layout.

    Returns:
        ``(data_dir, label_dir, basename)``.
    """
    root = tmp_path_factory.mktemp("mover_scene")
    data_dir, label_dir = root / "data_all", root / "label_msk"
    data_dir.mkdir()
    label_dir.mkdir()
    rng = np.random.default_rng(0)
    vol = rng.integers(0, 256, size=(64, 128, 128), dtype=np.uint8)
    mask = np.zeros((64, 128, 128), dtype=np.uint8)
    for t in range(64):  # a 2x2 mover drifting one pixel every 4 frames, plus a second one
        mask[t, 40:42, 10 + t // 4: 12 + t // 4] = 255
        mask[t, 97 + t % 3: 99 + t % 3, 77:79] = 255
    np.save(data_dir / "vid0_1.npy", vol)
    np.save(label_dir / "vid0_1_mask.npy", mask)
    return data_dir, label_dir, "vid0_1.npy"


def test_full_resolution_preserves_v0_5_contract(scene_dirs):
    """``output_resolution="full"`` returns the v0/v0.5 schema."""
    DATA_DIR, LABEL_DIR, name = scene_dirs
    ds = MoverDataset(DATA_DIR, LABEL_DIR, [name], output_resolution="full")
    sample = ds[0]
    assert set(sample.keys()) == {"volume", "mask", "file_name", "label"}
    assert sample["volume"].shape == (1, 64, 128, 128)
    assert sample["mask"].shape == (1, 64, 128, 128)
    assert sample["volume"].dtype == np.float32
    assert sample["mask"].dtype == np.float32


def test_low_res_returns_pooled_mask_and_full_mask(scene_dirs):
    """``output_resolution="low_res"`` returns ``mask`` downsampled and
    ``mask_full`` at original resolution."""
    DATA_DIR, LABEL_DIR, name = scene_dirs
    ds = MoverDataset(
        DATA_DIR, LABEL_DIR, [name], output_resolution="low_res",
    )
    sample = ds[0]
    # v0.8 §3.2 added the always-present "mask_hard" key for the metric
    # path. At soft_target=False (default), it is byte-identical to "mask".
    assert set(sample.keys()) == {
        "volume", "mask", "mask_hard", "mask_full", "file_name", "label",
    }
    assert sample["volume"].shape == (1, 64, 128, 128)
    assert sample["mask"].shape == (1, 32, 32, 32), (
        f"low-res mask should be (1, 32, 32, 32); got {sample['mask'].shape}"
    )
    assert sample["mask_hard"].shape == (1, 32, 32, 32)
    np.testing.assert_array_equal(sample["mask"], sample["mask_hard"])
    assert sample["mask_full"].shape == (1, 64, 128, 128)


def test_low_res_mask_matches_manual_pool(scene_dirs):
    """The dataset's downsampled mask matches a manual ``max_pool3d`` of
    the same mask with the same kernel/stride."""
    DATA_DIR, LABEL_DIR, name = scene_dirs
    ds = MoverDataset(
        DATA_DIR, LABEL_DIR, [name], output_resolution="low_res",
    )
    sample = ds[0]
    mask_full = torch.from_numpy(sample["mask_full"])  # (1, 64, 128, 128)
    manual_lr = F.max_pool3d(
        mask_full.unsqueeze(0),                         # (1, 1, T, H, W)
        kernel_size=(2, 4, 4), stride=(2, 4, 4),
    )[0]                                                # (1, 32, 32, 32)
    np.testing.assert_array_equal(sample["mask"], manual_lr.numpy())


def test_invalid_output_resolution_raises():
    """``output_resolution`` outside the valid set raises ``ValueError``."""
    with pytest.raises(ValueError):
        MoverDataset(
            "/nonexistent_dir", "/nonexistent_dir", [],
            output_resolution="medium",  # invalid
        )
