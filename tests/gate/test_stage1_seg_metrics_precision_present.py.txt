"""Phase 2 (§5 item 6) tests for the proposal-centric gate-precision
metrics ``precision_present`` and ``redundancy_present`` added to
``DetectionEpochAccumulator``.

These metrics are multi-target-safe (unlike ``precision@K``, whose
slot-denominator cannot cancel for N>1): they count *proposals that
contain a target* over *all present-scene proposals*, and *proposals per
distinct target hit*.

The accumulator's default patch (128x128) spans the whole 128-wide field,
so every proposal trivially contains every target at 128² — which is why
the metric only becomes informative at larger fields (512²/1024², the
Phase 2 regime). The tests therefore use small patches so proposals are
localized and a hand-built "2 of 3 proposals hit, 2 distinct targets
covered" case is constructible.
"""

from __future__ import annotations

import math
import sys
from pathlib import Path

import torch

_ROOT = Path(__file__).resolve().parents[2]

from localizer.gate.metrics import (  # noqa: E402
    DetectionEpochAccumulator,
)

# 128² field at the 1/8 rung => 16x16 patch (Phase 2 §5 item 7). With a
# full-field 128 patch the hit test passes trivially and tests nothing;
# 16x16 makes containment meaningful.
_SHAPE = (1, 8, 128, 128)
_PATCH = 16


def _logits(prob_at: list[tuple[int, int, int]],
            hot: float = 0.9, cold: float = 0.01) -> torch.Tensor:
    """Logits volume hitting ``hot`` at the given (t, h, w) voxels."""
    probs = torch.full(_SHAPE, cold)
    for t, h, w in prob_at:
        probs[0, t, h, w] = hot
    return torch.log(probs / (1.0 - probs))


def _mask(targets: list[tuple[int, int, int, int]]) -> torch.Tensor:
    """Binary mask with one 2x2 (h,w) footprint per (t_lo, t_hi, hc, wc)."""
    m = torch.zeros(_SHAPE)
    for (t_lo, t_hi, hc, wc) in targets:
        m[0, t_lo:t_hi, hc:hc + 2, wc:wc + 2] = 1.0
    return m


def test_precision_and_redundancy_present_handbuilt():
    """3 proposals, 2 contain a target, 2 distinct targets covered ->
    precision_present == 2/3, redundancy_present == 3/2."""
    # Three well-separated hotspots -> three connected components ->
    # three proposals. The 16x16 patch keeps them localized on the 128
    # field (each covers +/-8 around its centroid).
    acc = DetectionEpochAccumulator(ks=(1, 3, 5), patch_h=_PATCH, patch_w=_PATCH)
    logits = _logits([(4, 20, 20), (4, 20, 80), (4, 100, 100)])
    # Two targets, co-located with the first two hotspots; nothing near
    # the third hotspot at (100, 100), so that proposal is a pure FP.
    mask = _mask([(2, 6, 20, 20), (2, 6, 20, 80)])
    acc.update(logits, mask)

    out = acc.summarize()
    assert acc.proposals_present_total == 3
    assert acc.proposals_hit_total == 2
    assert acc.distinct_targets_hit_total == 2
    assert math.isclose(out["precision_present"], 2.0 / 3.0, rel_tol=1e-9)
    assert math.isclose(out["redundancy_present"], 3.0 / 2.0, rel_tol=1e-9)


def test_absent_scene_proposals_excluded():
    """Proposals on an absent (target-free) scene must not enter either
    the precision_present numerator/denominator or the redundancy stats."""
    acc = DetectionEpochAccumulator(ks=(1, 3, 5), patch_h=_PATCH, patch_w=_PATCH)
    # First, a present scene establishing precision_present = 2/3.
    acc.update(_logits([(4, 20, 20), (4, 20, 80), (4, 100, 100)]),
               _mask([(2, 6, 20, 20), (2, 6, 20, 80)]))
    # Then an absent scene with several hot proposals and an all-zero mask.
    acc.update(_logits([(4, 10, 10), (4, 60, 60), (4, 110, 110)]),
               torch.zeros(_SHAPE))

    out = acc.summarize()
    # Absent proposals counted only on the absent side.
    assert acc.n_absent == 1
    assert acc.proposals_absent_total == 3
    # Present-side gate-precision unchanged by the absent scene.
    assert acc.proposals_present_total == 3
    assert acc.proposals_hit_total == 2
    assert acc.distinct_targets_hit_total == 2
    assert math.isclose(out["precision_present"], 2.0 / 3.0, rel_tol=1e-9)
    assert math.isclose(out["redundancy_present"], 3.0 / 2.0, rel_tol=1e-9)


def test_cluster_recall_two_clusters_one_hit():
    """Two GRF clusters (separated chip groups), one covered by a
    proposal and one not -> cluster_recall == 0.5. Uses chip_size 32 on
    the 128 field (4x4 chip grid, matching the 128² dataset)."""
    acc = DetectionEpochAccumulator(ks=(1, 3, 5), patch_h=_PATCH, patch_w=_PATCH,
                                    cluster_chip_size=32)
    # Cluster A around chip (0,0): target at (20,20); a proposal hits it.
    # Cluster B around chip (3,3): target at (110,110); NO proposal there.
    # (chips (0,0) and (3,3) are not 4-connected -> two clusters.)
    logits = _logits([(4, 20, 20)])  # one hotspot -> one proposal on cluster A
    # t_lo=0 so targets are present at the frame-0 placement frame, which
    # is what the chip-occupancy / cluster CC test reads (real data places
    # all movers at frame 0).
    mask = _mask([(0, 6, 20, 20), (0, 6, 110, 110)])
    acc.update(logits, mask)
    out = acc.summarize()
    assert acc.clusters_total == 2
    assert acc.clusters_hit_total == 1
    assert math.isclose(out["cluster_recall"], 0.5, rel_tol=1e-9)


def test_cluster_recall_disabled_is_nan():
    """Without cluster_chip_size, cluster_recall is NaN and the counters
    stay zero (default-off, no behavior change)."""
    acc = DetectionEpochAccumulator(ks=(1, 3, 5), patch_h=_PATCH, patch_w=_PATCH)
    acc.update(_logits([(4, 20, 20)]), _mask([(2, 6, 20, 20)]))
    out = acc.summarize()
    assert acc.clusters_total == 0
    assert math.isnan(out["cluster_recall"])


def test_no_present_proposals_gives_nan():
    """Cold-start (predict-nothing) present scene -> 0 proposals ->
    precision_present and redundancy_present are NaN (guarded)."""
    acc = DetectionEpochAccumulator(ks=(1, 3, 5), patch_h=_PATCH, patch_w=_PATCH)
    # All-cold logits -> no thresholded voxels -> no proposals; one target.
    acc.update(_logits([]), _mask([(2, 6, 20, 20)]))
    out = acc.summarize()
    assert acc.proposals_present_total == 0
    assert math.isnan(out["precision_present"])
    assert math.isnan(out["redundancy_present"])
