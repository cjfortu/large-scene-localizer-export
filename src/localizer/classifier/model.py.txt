"""Mover3DCNNTrans binary scene classifier (recovered from earlier Stage 2 work).

3-stage 3D CNN encoder (1 -> 32 -> 64 -> 128, ``Conv3d 3x3x3 + BN + ReLU +
MaxPool3d(2)`` per stage) -> single ``RelativeAttentionBlock`` -> MLP head
with two Dropout(0.5) layers -> single logit -> sigmoid (in ``forward``).

Recovery adaptations for native 64-frame input (vs. the original engineer's
32-frame input after upstream 144x192x48 -> 128x128x48 resize + further
downsampling):

* Classifier-head Linear input dimension parameterized via ``t_dim``: the
  encoder applies three 2x2x2 max-pools, so the flattened bottleneck has
  shape ``(128, t_dim // 8, 16, 16)``. For ``t_dim = 64`` that's 8 temporal
  cells; the original used 4. The spatial ``16 * 16`` assumes 128x128 input.
* ``max_attn_len`` passed through to the ``RelativeAttentionBlock`` so the
  relative-bias table accommodates the larger token count.

The 16x16 spatial assumption holds for the recovered task (128x128 input);
it would need to be updated if the input spatial dimension changes.
"""

import torch
import torch.nn as nn

from localizer.classifier.relative_attention import RelativeAttentionBlock
from localizer.common.swin3d import SwinTransformerBlock3d


class _SwinAttnAdapter(nn.Module):
    """Paired 3D Swin blocks adapter, shape-preserving on ``(B, C, T, H, W)``.

    Wraps two ``SwinTransformerBlock3d`` blocks (block 0 unshifted, block 1
    shifted by half-window) and handles the permute glue from Conv3d's
    channel-first ``(B, C, T, H, W)`` to Swin's channels-last
    ``(B, T, H, W, C)`` and back. Mirrors the segmenter's ``Bottleneck``
    pattern (``stage2_volseg/model.py``) including the trailing
    ``LayerNorm(C)`` before the permute-back.

    Args:
        dim: Channel dimension (128 for the classifier bottleneck).
        num_heads: Attention heads.
        window_size: 3-tuple ``(T, H, W)``. Half-window shift for block 1
            is computed as ``(w0//2, w1//2, w2//2)``.
        attn_drop: Dropout after softmax in the windowed attention.
        proj_drop: Dropout after the windowed-attention output projection.
    """

    def __init__(
        self,
        dim: int = 128,
        num_heads: int = 4,
        window_size: tuple[int, int, int] = (2, 4, 4),
        attn_drop: float = 0.0,
        proj_drop: float = 0.0,
    ):
        super().__init__()
        shift = (window_size[0] // 2, window_size[1] // 2, window_size[2] // 2)
        self.blocks = nn.ModuleList([
            SwinTransformerBlock3d(
                dim=dim,
                num_heads=num_heads,
                window_size=window_size,
                shift_size=(0, 0, 0) if (i % 2 == 0) else shift,
                attn_drop=attn_drop,
                proj_drop=proj_drop,
            )
            for i in range(2)
        ])
        self.norm = nn.LayerNorm(dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = x.permute(0, 2, 3, 4, 1)  # (B, T, H, W, C)
        for blk in self.blocks:
            x = blk(x)
        x = self.norm(x)
        x = x.permute(0, 4, 1, 2, 3)  # (B, C, T, H, W)
        return x


class _SwinAttnAdapterWithPatchEmbed(nn.Module):
    """``_SwinAttnAdapter`` wrapped with a Conv3d(C, C, 1) patch-embed.

    Adds the segmenter ``Bottleneck``'s 1x1x1 channel-projection step
    before the permute-to-channels-last + paired Swin blocks. Isolates
    the patch-embed difference (#1 of the five segmenter/classifier
    differences) at the classifier's smaller scale (dim=128).
    """

    def __init__(
        self,
        dim: int = 128,
        num_heads: int = 4,
        window_size: tuple[int, int, int] = (2, 4, 4),
        attn_drop: float = 0.0,
        proj_drop: float = 0.0,
    ):
        super().__init__()
        self.patch_embed = nn.Conv3d(dim, dim, kernel_size=1, stride=1)
        self.swin = _SwinAttnAdapter(
            dim=dim,
            num_heads=num_heads,
            window_size=window_size,
            attn_drop=attn_drop,
            proj_drop=proj_drop,
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.patch_embed(x)
        return self.swin(x)


class _SwinAttnAdapterPostNorm(nn.Module):
    """``_SwinAttnAdapter`` rewired to post-norm.

    Same paired Swin blocks as ``_SwinAttnAdapter`` but with
    LayerNorm applied AFTER the attn residual and AFTER the MLP residual
    instead of before. Tests whether pre-norm placement (a known
    gradient-attenuation regime when residuals dominate) is the failure
    mode in the as-shipped configuration.
    """

    def __init__(
        self,
        dim: int = 128,
        num_heads: int = 4,
        window_size: tuple[int, int, int] = (2, 4, 4),
        attn_drop: float = 0.0,
        proj_drop: float = 0.0,
    ):
        super().__init__()
        shift = (window_size[0] // 2, window_size[1] // 2, window_size[2] // 2)
        self.blocks = nn.ModuleList([
            SwinTransformerBlock3d(
                dim=dim,
                num_heads=num_heads,
                window_size=window_size,
                shift_size=(0, 0, 0) if (i % 2 == 0) else shift,
                attn_drop=attn_drop,
                proj_drop=proj_drop,
                norm_position="post",
            )
            for i in range(2)
        ])
        self.norm = nn.LayerNorm(dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = x.permute(0, 2, 3, 4, 1)
        for blk in self.blocks:
            x = blk(x)
        x = self.norm(x)
        x = x.permute(0, 4, 1, 2, 3)
        return x


class _SwinAttnAdapterWindow444(nn.Module):
    """``_SwinAttnAdapter`` with ``window_size=(4, 4, 4)`` instead of ``(2, 4, 4)``.

    Doubles the temporal extent of the windowed attention; no other
    changes. Tests whether the temporal-window size is the failure
    knob.
    """

    def __init__(
        self,
        dim: int = 128,
        num_heads: int = 4,
        attn_drop: float = 0.0,
        proj_drop: float = 0.0,
    ):
        super().__init__()
        window_size = (4, 4, 4)
        shift = (window_size[0] // 2, window_size[1] // 2, window_size[2] // 2)
        self.blocks = nn.ModuleList([
            SwinTransformerBlock3d(
                dim=dim,
                num_heads=num_heads,
                window_size=window_size,
                shift_size=(0, 0, 0) if (i % 2 == 0) else shift,
                attn_drop=attn_drop,
                proj_drop=proj_drop,
            )
            for i in range(2)
        ])
        self.norm = nn.LayerNorm(dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = x.permute(0, 2, 3, 4, 1)
        for blk in self.blocks:
            x = blk(x)
        x = self.norm(x)
        x = x.permute(0, 4, 1, 2, 3)
        return x


class _SwinAttnAdapterSegmenterWiring(nn.Module):
    """Full segmenter ``Bottleneck`` wiring at the classifier's dim=128 scale.

    Applies four of the five segmenter/classifier differences in one
    variant: (#1) Conv3d(C, C, 1) patch-embed before the paired Swin
    blocks; (#2) ``attn_drop=0``; (#3) ``proj_drop=0``; (#4) ``drop_path``
    linspace ``[0, 0.2]`` across the two paired blocks. Pre-norm is
    preserved (the segmenter uses pre-norm). Difference #5 (scale —
    dim=128 here vs dim=512 in the segmenter) is held fixed.

    The classifier config's ``attn_dropout`` field is ignored here:
    segmenter wiring zeros both attn_drop and proj_drop as part of the
    four-variable change.
    """

    def __init__(
        self,
        dim: int = 128,
        num_heads: int = 4,
        window_size: tuple[int, int, int] = (2, 4, 4),
    ):
        super().__init__()
        self.patch_embed = nn.Conv3d(dim, dim, kernel_size=1, stride=1)
        shift = (window_size[0] // 2, window_size[1] // 2, window_size[2] // 2)
        dpr = [v.item() for v in torch.linspace(0, 0.2, 2)]
        self.blocks = nn.ModuleList([
            SwinTransformerBlock3d(
                dim=dim,
                num_heads=num_heads,
                window_size=window_size,
                shift_size=(0, 0, 0) if (i % 2 == 0) else shift,
                attn_drop=0.0,
                proj_drop=0.0,
                drop_path=dpr[i],
            )
            for i in range(2)
        ])
        self.norm = nn.LayerNorm(dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.patch_embed(x)
        x = x.permute(0, 2, 3, 4, 1)
        for blk in self.blocks:
            x = blk(x)
        x = self.norm(x)
        x = x.permute(0, 4, 1, 2, 3)
        return x


class _ChannelTokenAttn(nn.Module):
    """Channel-as-token attention block with a projection bottleneck.

    Shape-preserving on ``(B, C, T, H, W)``. Tokenization is the transpose of
    ``RelativeAttentionBlock``: each of the ``C = 128`` channels becomes a
    token, and the flattened spatial-temporal extent ``T*H*W`` (= 2048 for
    ``t_dim=64``) is that token's embedding. Tests the "which feature
    channels are relevant" inductive bias instead of "which positions are
    relevant."

    A raw 2048-dim embedding would be near-degenerate (almost all capacity
    in projection matrices, not in token mixing), so this variant projects
    each channel-token down to ``proj_dim`` (default 512) before attention,
    runs a pre-norm MHA + FFN block at that width, then projects back up to
    the spatial-temporal extent. ``proj_dim`` is a **named hyperparameter**
    of this variant — the ``channel_token`` result reflects this choice, not
    only the channels-as-tokens idea.

    Deliberate design choice: **no relative-position bias**. Channel
    ordering in the conv encoder's output is arbitrary (there is no
    meaningful "distance" between channel 5 and channel 90), so a 1-D
    relative-position bias indexed by token distance would inject spurious
    structure. Plain scaled-dot-product attention only, via
    ``nn.MultiheadAttention``.

    Args:
        t_dim: Number of input frames T. Used to size the spatial-temporal
            extent as ``(t_dim // 8) * 16 * 16`` (three 2x2x2 max-pools in
            the encoder; 16*16 spatial assumes 128x128 input — same
            reasoning as the flatten-head Linear).
        proj_dim: Projection-bottleneck width. Must be divisible by
            ``num_heads``. Default 512.
        num_heads: Attention heads.
        dropout: Dropout for attention output, FFN, and the MHA module.
    """

    def __init__(
        self,
        t_dim: int = 64,
        proj_dim: int = 512,
        num_heads: int = 4,
        dropout: float = 0.1,
    ):
        super().__init__()
        assert proj_dim % num_heads == 0, (
            f"proj_dim {proj_dim} must be divisible by num_heads {num_heads}"
        )
        self.spatial = (t_dim // 8) * 16 * 16
        self.proj_dim = proj_dim

        self.down_proj = nn.Linear(self.spatial, proj_dim)
        self.up_proj = nn.Linear(proj_dim, self.spatial)

        self.norm1 = nn.LayerNorm(proj_dim)
        self.norm2 = nn.LayerNorm(proj_dim)
        self.attn = nn.MultiheadAttention(
            embed_dim=proj_dim,
            num_heads=num_heads,
            dropout=dropout,
            batch_first=True,
        )
        self.attn_drop = nn.Dropout(dropout)
        self.ffn = nn.Sequential(
            nn.Linear(proj_dim, proj_dim * 4),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(proj_dim * 4, proj_dim),
            nn.Dropout(dropout),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        B, C, T, H, W = x.shape
        assert T * H * W == self.spatial, (
            f"channel_token: T*H*W ({T*H*W}) != construction-time spatial "
            f"({self.spatial}); was the module built with the wrong t_dim?"
        )
        # Channels as tokens: (B, C, T*H*W) — no .transpose(1, 2).
        tokens = x.view(B, C, T * H * W)
        tokens = self.down_proj(tokens)  # (B, C, proj_dim)

        residual = tokens
        h = self.norm1(tokens)
        h, _ = self.attn(h, h, h, need_weights=False)
        tokens = residual + self.attn_drop(h)

        residual = tokens
        h = self.norm2(tokens)
        tokens = residual + self.ffn(h)

        tokens = self.up_proj(tokens)  # (B, C, spatial)
        return tokens.view(B, C, T, H, W)


class Mover3DCNNTrans(nn.Module):
    """Binary scene classifier: 3D CNN + token-mixing block + MLP head.

    Args:
        t_dim: Number of input frames T (default 64). Determines the
            classifier-head Linear input dim via ``t_dim // 8`` (only when
            ``head_type='flatten'``; the ``'gap'`` head pools spatially).
        dropout_rate: Dropout in the classifier-head MLP.
        attn_dropout: Dropout in the token-mixing block. For ``attn_type='full'``
            this is the ``RelativeAttentionBlock`` dropout; for ``'swin'`` it is
            both ``attn_drop`` and ``proj_drop`` on the paired Swin blocks;
            for ``'channel_token'`` it is applied inside ``nn.MultiheadAttention``,
            on the attention-output residual, and within the FFN; ignored for
            ``'none'``.
        max_attn_len: ``max_len`` argument for the ``'full'`` attention block's
            relative-bias table. Must be >= ``(t_dim // 8) * 16 * 16``. Ignored
            for ``attn_type='none'``, ``'swin'`` (Swin has its own
            window-sized relative-position bias), and ``'channel_token'`` (no
            relative-position bias — channel ordering is arbitrary).
        head_type: Classifier head variant. ``'flatten'`` (default) is the
            original ``Flatten + Linear(128 * (t_dim//8) * 16 * 16, 64)``
            head (~16.78M params for ``t_dim=64``). ``'gap'`` substitutes
            ``AdaptiveAvgPool3d((1,1,1)) + Flatten + Linear(128, 64)``
            (~8K params) to reduce head over-capacity.
        return_logits: If ``True``, ``forward`` returns raw pre-sigmoid
            logits (use with ``BCEWithLogitsLoss``). If ``False`` (default),
            ``forward`` applies sigmoid (use with ``BCELoss``).
        attn_type: Token-mixing block variant inserted between the 3D CNN
            encoder and the classifier head. ``'full'`` (default) uses the
            existing ``RelativeAttentionBlock`` (global attention over all
            ``(t_dim//8) * 16 * 16`` tokens) — the control. ``'none'``
            replaces the entire transformer block with ``nn.Identity()``
            (no attention sub-layer, no FFN, no LayerNorms, no residuals).
            ``'swin'`` uses paired ``SwinTransformerBlock3d`` blocks (depth
            2, window ``(2,4,4)``, block 0 unshifted, block 1 shifted by
            half-window) — local windowed attention with a cross-window
            information path via the shifted block. ``'channel_token'`` is
            the transpose tokenization: each of the 128 channels becomes a
            token and the flattened spatial-temporal extent ``T*H*W`` is
            that token's embedding, with a **512-dim projection bottleneck**
            (named hyperparameter of this variant) before a pre-norm MHA +
            FFN block, then projected back up. **No relative-position bias**
            — channel ordering is arbitrary, so position-distance bias would
            inject spurious structure.
    """

    def __init__(
        self,
        t_dim: int = 64,
        dropout_rate: float = 0.5,
        attn_dropout: float = 0.1,
        max_attn_len: int = 2200,
        head_type: str = "flatten",
        return_logits: bool = False,
        attn_type: str = "full",
    ):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv3d(1, 32, kernel_size=(3, 3, 3), padding="same"),
            nn.BatchNorm3d(32),
            nn.ReLU(),
            nn.MaxPool3d(kernel_size=2),

            nn.Conv3d(32, 64, kernel_size=(3, 3, 3), padding="same"),
            nn.BatchNorm3d(64),
            nn.ReLU(),
            nn.MaxPool3d(kernel_size=2),

            nn.Conv3d(64, 128, kernel_size=(3, 3, 3), padding="same"),
            nn.BatchNorm3d(128),
            nn.ReLU(),
            nn.MaxPool3d(kernel_size=2),  # (t_dim//8, 16, 16)
        )

        if attn_type == "full":
            self.attn = RelativeAttentionBlock(
                dim=128, num_heads=4, max_len=max_attn_len, dropout=attn_dropout,
            )
        elif attn_type == "none":
            self.attn = nn.Identity()
        elif attn_type == "swin":
            self.attn = _SwinAttnAdapter(
                dim=128,
                num_heads=4,
                window_size=(2, 4, 4),
                attn_drop=attn_dropout,
                proj_drop=attn_dropout,
            )
        elif attn_type == "channel_token":
            self.attn = _ChannelTokenAttn(
                t_dim=t_dim,
                proj_dim=512,
                num_heads=4,
                dropout=attn_dropout,
            )
        elif attn_type == "swin_patch_embed":
            self.attn = _SwinAttnAdapterWithPatchEmbed(
                dim=128,
                num_heads=4,
                window_size=(2, 4, 4),
                attn_drop=attn_dropout,
                proj_drop=attn_dropout,
            )
        elif attn_type == "swin_postnorm":
            self.attn = _SwinAttnAdapterPostNorm(
                dim=128,
                num_heads=4,
                window_size=(2, 4, 4),
                attn_drop=attn_dropout,
                proj_drop=attn_dropout,
            )
        elif attn_type == "swin_window444":
            self.attn = _SwinAttnAdapterWindow444(
                dim=128,
                num_heads=4,
                attn_drop=attn_dropout,
                proj_drop=attn_dropout,
            )
        elif attn_type == "swin_segmenter_wiring":
            self.attn = _SwinAttnAdapterSegmenterWiring(
                dim=128,
                num_heads=4,
                window_size=(2, 4, 4),
            )
        else:
            raise ValueError(
                f"Unknown attn_type: {attn_type!r} "
                "(expected 'full', 'none', 'swin', 'channel_token', "
                "'swin_patch_embed', 'swin_postnorm', 'swin_window444', "
                "or 'swin_segmenter_wiring')"
            )

        # Original flatten head (default): 16.78M params for t_dim=64.
        # GAP head alternative: ~8K params (head_in_dim = 128 channels only).
        if head_type == "flatten":
            head_in_dim = 128 * (t_dim // 8) * 16 * 16
            self.head_prefix = nn.Flatten()
        elif head_type == "gap":
            head_in_dim = 128
            self.head_prefix = nn.Sequential(
                nn.AdaptiveAvgPool3d((1, 1, 1)),
                nn.Flatten(),
            )
        else:
            raise ValueError(f"Unknown head_type: {head_type!r} (expected 'flatten' or 'gap')")

        self.classifier = nn.Sequential(
            nn.Linear(head_in_dim, 64),
            nn.BatchNorm1d(64),
            nn.ReLU(),
            nn.Dropout(dropout_rate),

            nn.Linear(64, 32),
            nn.BatchNorm1d(32),
            nn.ReLU(),
            nn.Dropout(dropout_rate),

            nn.Linear(32, 1),
        )

        self.return_logits = return_logits

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x = self.features(x)
        x = self.attn(x)
        x = self.head_prefix(x)
        x = self.classifier(x)
        if self.return_logits:
            return x.squeeze(1)
        return torch.sigmoid(x).squeeze(1)
