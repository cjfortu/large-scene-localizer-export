"""3D Swin transformer block (recovered from earlier Stage 2 work).

Derived from the official PyTorch Swin implementation. Adapted for 3D inputs
(T, H, W) and to accept Conv3d feature outputs at the bottleneck (no spatial
patch-embedding inside this module — that happens in ``model.Bottleneck``'s
1x1x1 patch-embed convolution).

There is no attention mask: the bottleneck attends to all tokens within
each window, regardless of padding. Inputs at the bottleneck are
feature-oriented (Conv3d outputs), not raw-spatial — all tokens carry signal.

Recovery note: the original ``SwinTransformer3d_mod.py`` set module-level
torch / numpy / random seeds on import. That has been removed here. The
training entry point (``train.py``) is responsible for seeding before any
model construction. Module-import-time seed-setting (a) is order-dependent,
(b) breaks DataLoader-worker determinism, and (c) conflicts with the
project convention that seeds are set explicitly at entry points.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.nn.init as init


class DropPath(nn.Module):
    """Stochastic depth — drop the residual path with probability ``drop_prob``."""

    def __init__(self, drop_prob: float = 0.0):
        super().__init__()
        self.drop_prob = drop_prob

    def _drop_path(self, x):
        if self.drop_prob == 0. or not self.training:
            return x
        keep_prob = 1 - self.drop_prob
        shape = (x.shape[0],) + (1,) * (x.ndim - 1)
        random_tensor = keep_prob + torch.rand(shape, dtype=x.dtype, device=x.device)
        random_tensor.floor_()
        output = x.div(keep_prob) * random_tensor
        return output

    def forward(self, x):
        return self._drop_path(x)


class Mlp(nn.Module):
    """Standard FFN block: Linear -> GELU -> Dropout -> Linear -> Dropout."""

    def __init__(self, in_features, hidden_features=None, out_features=None, drop=0.):
        super().__init__()
        out_features = out_features or in_features
        hidden_features = hidden_features or in_features
        self.fc1 = nn.Linear(in_features, hidden_features)
        self.act = nn.GELU()
        self.fc2 = nn.Linear(hidden_features, out_features)
        self.drop = nn.Dropout(drop)

    def forward(self, x):
        x = self.fc1(x)
        x = self.act(x)
        x = self.drop(x)
        x = self.fc2(x)
        x = self.drop(x)
        return x


class WindowAttention3d(nn.Module):
    """Windowed multi-head self-attention with 3D relative position bias.

    Args:
        dim: Embedding dimension.
        window_size: 3-tuple ``(T, H, W)``.
        num_heads: Attention heads.
        attn_drop: Dropout after softmax.
        proj_drop: Dropout after the output projection.
    """

    def __init__(self, dim, window_size, num_heads, attn_drop=0., proj_drop=0.):
        super().__init__()
        self.dim = dim
        self.window_size = window_size
        self.num_heads = num_heads
        head_dim = dim // num_heads
        self.scale = head_dim ** -0.5

        self.relative_position_bias_table = nn.Parameter(
            torch.zeros(
                (2 * window_size[0] - 1) * (2 * window_size[1] - 1) * (2 * window_size[2] - 1),
                num_heads,
            )
        )
        init.trunc_normal_(self.relative_position_bias_table, std=0.02)

        coords_dhw = [torch.arange(self.window_size[i]) for i in range(3)]
        coords = torch.stack(
            torch.meshgrid(coords_dhw[0], coords_dhw[1], coords_dhw[2], indexing="ij")
        )  # 3, Wd, Wh, Ww
        coords_flatten = torch.flatten(coords, 1)
        relative_coords = coords_flatten[:, :, None] - coords_flatten[:, None, :]
        relative_coords = relative_coords.permute(1, 2, 0).contiguous()
        relative_coords[:, :, 0] += self.window_size[0] - 1
        relative_coords[:, :, 1] += self.window_size[1] - 1
        relative_coords[:, :, 2] += self.window_size[2] - 1
        relative_coords[:, :, 0] *= (2 * self.window_size[1] - 1) * (2 * self.window_size[2] - 1)
        relative_coords[:, :, 1] *= (2 * self.window_size[2] - 1)
        self.relative_posn_idx = relative_coords.sum(dim=-1)
        self.register_buffer(
            "relative_position_index", self.relative_posn_idx.view(-1).to(torch.long)
        )

        self.qkv = nn.Linear(dim, dim * 3, bias=True)
        self.attn_drop = nn.Dropout(attn_drop)
        self.proj = nn.Linear(dim, dim)
        self.proj_drop = nn.Dropout(proj_drop)
        self.softmax = nn.Softmax(dim=-1)

    def forward(self, x):
        B_, N, C = x.shape
        qkv = self.qkv(x).reshape(B_, N, 3, self.num_heads, C // self.num_heads).permute(
            2, 0, 3, 1, 4
        )
        q, k, v = qkv[0], qkv[1], qkv[2]
        q = q * self.scale
        attn = q @ k.transpose(-2, -1)

        window_vol = self.window_size[0] * self.window_size[1] * self.window_size[2]
        relative_position_bias = self.relative_position_bias_table[
            self.relative_posn_idx[:window_vol, :window_vol].flatten()
        ]
        relative_position_bias = relative_position_bias.view(window_vol, window_vol, -1)
        relative_position_bias = relative_position_bias.permute(2, 0, 1).contiguous().unsqueeze(0)

        attn = self.softmax(attn + relative_position_bias)
        attn = self.attn_drop(attn)
        x = (attn @ v).transpose(1, 2).reshape(B_, N, C)
        x = self.proj(x)
        x = self.proj_drop(x)
        return x


class SwinTransformerBlock3d(nn.Module):
    """3D Swin transformer block: windowed (optionally shifted) MHA + MLP.

    Inputs are channels-last ``(B, D, H, W, C)`` (the caller permutes from
    Conv3d's ``(B, C, T, H, W)`` before this block and back after).
    """

    def __init__(
        self,
        dim,
        num_heads,
        window_size=(2, 4, 4),
        shift_size=(0, 0, 0),
        mlp_ratio=4.,
        proj_drop=0.,
        attn_drop=0.,
        ffn_drop=0.,
        drop_path=0.,
        norm_position="pre",
    ):
        super().__init__()
        if norm_position not in ("pre", "post"):
            raise ValueError(
                f"norm_position must be 'pre' or 'post', got {norm_position!r}"
            )
        self.dim = dim
        self.num_heads = num_heads
        self.window_size = window_size
        self.shift_size = shift_size
        self.mlp_ratio = mlp_ratio
        self.norm_position = norm_position
        self.norm1 = nn.LayerNorm(dim)
        self.attn = WindowAttention3d(
            dim,
            window_size=self.window_size,
            num_heads=num_heads,
            attn_drop=attn_drop,
            proj_drop=proj_drop,
        )
        self.drop_path = DropPath(drop_path) if drop_path > 0. else nn.Identity()
        self.norm2 = nn.LayerNorm(dim)
        mlp_hidden_dim = int(dim * mlp_ratio)
        self.mlp = Mlp(in_features=dim, hidden_features=mlp_hidden_dim, drop=ffn_drop)

    def _window_partition3d(self, x):
        B, D, H, W, C = x.shape
        winsz = self.window_size
        x = x.view(B, D // winsz[0], winsz[0], H // winsz[1], winsz[1], W // winsz[2], winsz[2], C)
        windows = (
            x.permute(0, 1, 3, 5, 2, 4, 6, 7)
            .contiguous()
            .view(-1, winsz[0] * winsz[1] * winsz[2], C)
        )
        return windows

    def _window_reverse3d(self, windows, B, D, H, W):
        winsz = self.window_size
        x = windows.view(
            B, D // winsz[0], H // winsz[1], W // winsz[2], winsz[0], winsz[1], winsz[2], -1
        )
        x = x.permute(0, 1, 4, 2, 5, 3, 6, 7).contiguous().view(B, D, H, W, -1)
        return x

    def forward(self, x):
        B, D, H, W, C = x.shape
        shortcut = x
        if self.norm_position == "pre":
            x = self.norm1(x)

        dhw = (D, H, W)
        winsz = self.window_size
        pad_l = pad_t = pad_d0 = 0
        pad_d1, pad_b, pad_r = [(winsz[i] - dhw[i] % winsz[i]) % winsz[i] for i in range(3)]
        x = F.pad(x, (0, 0, pad_l, pad_r, pad_t, pad_b, pad_d0, pad_d1))
        _, Dp, Hp, Wp, _ = x.shape

        if any(i > 0 for i in self.shift_size):
            shifted_x = torch.roll(
                x,
                shifts=(-self.shift_size[0], -self.shift_size[1], -self.shift_size[2]),
                dims=(1, 2, 3),
            )
        else:
            shifted_x = x
        x_windows = self._window_partition3d(shifted_x)

        attn_windows = self.attn(x_windows)
        shifted_x = self._window_reverse3d(attn_windows, B, Dp, Hp, Wp)

        if any(i > 0 for i in self.shift_size):
            x = torch.roll(
                shifted_x,
                shifts=(self.shift_size[0], self.shift_size[1], self.shift_size[2]),
                dims=(1, 2, 3),
            )
        else:
            x = shifted_x

        if pad_d1 > 0 or pad_b > 0 or pad_r > 0:
            x = x[:, :D, :H, :W, :].contiguous()

        if self.norm_position == "pre":
            x = shortcut + self.drop_path(x)
            x = x + self.drop_path(self.mlp(self.norm2(x)))
        else:
            x = self.norm1(shortcut + self.drop_path(x))
            x = self.norm2(x + self.drop_path(self.mlp(x)))
        return x
